from pyspark.sql import SparkSession
from pyspark.sql.functions import col, udf
from datetime import datetime, timedelta

# Initialize Spark Session
spark = SparkSession.builder \
    .appName("ResumePipeline") \
    .config("spark.sql.adaptive.enabled", "true") \
    .getOrCreate()

# Bronze Layer — Raw Ingestion
df_raw = spark.read \
    .format("json") \
    .option("multiLine", "true") \
    .load("s3://data-lake/bronze/profile/*.json")

# Silver Layer — Clean & Transform
df_clean = df_raw \
    .filter(col("experience_years") >= 1) \
    .dropDuplicates(["company", "role"]) \
    .withColumn("processed_at", current_timestamp()) \
    .orderBy(col("start_date").desc())

# Gold Layer — Aggregate Insights
df_gold = df_clean.groupBy("skill_category") \
    .agg(
        count("*").alias("skill_count"),
        collect_list("technology").alias("stack")
    ) \
    .orderBy(col("skill_count").desc())

# Write to curated store
df_gold.write \
    .format("delta") \
    .mode("overwrite") \
    .save("s3://data-lake/gold/insights/")

print(f"✅ Pipeline complete — {
      df_gold.count()} categories processed")
